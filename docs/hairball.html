A: ```python
from pathlib import Path

def safe_upload_path(upload_dir: str, filename: str) -> str:
    upload_path = Path(upload_dir).expanduser().resolve(strict=False)
    if Path(filename).is_absolute():
        raise ValueError("Absolute filenames are not allowed")
    candidate = (upload_path / filename).resolve(strict=False)
    if candidate.parts[:len(upload_path.parts)] != upload_path.parts:
        raise ValueError("Path traversal detected")
    return str(candidate)
```
